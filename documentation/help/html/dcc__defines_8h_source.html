115// =============================================================================
116// Auto-refresh pacing (command station scheduler, issue #5)
117//
118// A refresh slot is sent at full rate for DCC_REFRESH_PROMPT_SENDS turns after
119// each insert, then drops to a keep-alive: once every DCC_REFRESH_COLD_CYCLES
120// packet cycles, and never less often than every DCC_REFRESH_COLD_MAX_CYCLES.
121// Counts are packet cycles (one packet on the track each, about 6-7 ms for a
122// speed packet; longer packets make a cycle longer). Each can be overridden in
123// dcc_user_config.h.
124//
125// The ceiling is what a decoder's packet time-out (CV11, S-9.2.4 sec 4) has to
126// cover for an idle locomotive: at ~6.7 ms per cycle the default is ~0.8 s,
127// plus one packet per other slot that falls overdue on the same cycle. Keep
128// CV11 on such decoders at 0 (off) or at least DCC_REFRESH_CV11_FLOOR.
129// =============================================================================
130
134#ifndef DCC_REFRESH_PROMPT_SENDS
135#define DCC_REFRESH_PROMPT_SENDS 3
136#endif
137
140#ifndef DCC_REFRESH_COLD_CYCLES
141#define DCC_REFRESH_COLD_CYCLES 60
142#endif
143
150#ifndef DCC_REFRESH_COLD_MAX_CYCLES
151#define DCC_REFRESH_COLD_MAX_CYCLES 120
152#endif
153
160#ifndef DCC_REFRESH_CV11_FLOOR
161#define DCC_REFRESH_CV11_FLOOR 20
162#endif
163
164#if DCC_REFRESH_COLD_CYCLES > 0 && DCC_REFRESH_COLD_MAX_CYCLES < DCC_REFRESH_COLD_CYCLES
165#error "DCC_REFRESH_COLD_MAX_CYCLES must be >= DCC_REFRESH_COLD_CYCLES"
166#endif
167#if DCC_REFRESH_COLD_CYCLES > 0 && DCC_REFRESH_PROMPT_SENDS < 1
168#error "DCC_REFRESH_PROMPT_SENDS must be >= 1 while the cold tier is on, or a change waits a whole keep-alive interval"
169#endif
170#if DCC_REFRESH_COLD_MAX_CYCLES > 65535 || DCC_REFRESH_PROMPT_SENDS > 255
171#error "DCC_REFRESH_COLD_MAX_CYCLES must fit 16 bits and DCC_REFRESH_PROMPT_SENDS 8 bits"
172#endif
