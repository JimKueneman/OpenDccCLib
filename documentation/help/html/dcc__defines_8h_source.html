118// =============================================================================
119// Auto-refresh pacing (command station scheduler, issue #5)
120//
121// A refresh slot is sent at full rate for DCC_REFRESH_PROMPT_SENDS turns after
122// each insert, then drops to a keep-alive: once every DCC_REFRESH_COLD_CYCLES
123// packet cycles, and never less often than every DCC_REFRESH_COLD_MAX_CYCLES.
124// Counts are packet cycles (one packet on the track each, about 6-7 ms for a
125// speed packet; longer packets make a cycle longer). Each can be overridden in
126// dcc_user_config.h.
127//
128// The ceiling is what a decoder's packet time-out (CV11, S-9.2.4 sec 4) has to
129// cover for an idle locomotive: at ~6.7 ms per cycle the default is ~0.8 s,
130// plus one packet per other slot that falls overdue on the same cycle. Keep
131// CV11 on such decoders at 0 (off) or at least DCC_REFRESH_CV11_FLOOR.
132// =============================================================================
133
137#ifndef DCC_REFRESH_PROMPT_SENDS
138#define DCC_REFRESH_PROMPT_SENDS 3
139#endif
140
143#ifndef DCC_REFRESH_COLD_CYCLES
144#define DCC_REFRESH_COLD_CYCLES 60
145#endif
146
153#ifndef DCC_REFRESH_COLD_MAX_CYCLES
154#define DCC_REFRESH_COLD_MAX_CYCLES 120
155#endif
156
163#ifndef DCC_REFRESH_CV11_FLOOR
164#define DCC_REFRESH_CV11_FLOOR 20
165#endif
166
167#if DCC_REFRESH_COLD_CYCLES > 0 && DCC_REFRESH_COLD_MAX_CYCLES < DCC_REFRESH_COLD_CYCLES
168#error "DCC_REFRESH_COLD_MAX_CYCLES must be >= DCC_REFRESH_COLD_CYCLES"
169#endif
170#if DCC_REFRESH_COLD_CYCLES > 0 && DCC_REFRESH_PROMPT_SENDS < 1
171#error "DCC_REFRESH_PROMPT_SENDS must be >= 1 while the cold tier is on, or a change waits a whole keep-alive interval"
172#endif
173#if DCC_REFRESH_COLD_MAX_CYCLES > 65535 || DCC_REFRESH_PROMPT_SENDS > 255
174#error "DCC_REFRESH_COLD_MAX_CYCLES must fit 16 bits and DCC_REFRESH_PROMPT_SENDS 8 bits"
175#endif
